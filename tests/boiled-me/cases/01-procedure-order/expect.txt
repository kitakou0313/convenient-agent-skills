# 順序依存の手順。順序が崩れると手順が成立しない
must: touch /var/run/maintenance.flag
must: pg_dump -Fc app_db
must: backup_0301.dump
must: migrate --plan
must: systemctl restart app.service
must: rm /var/run/maintenance.flag
order: touch /var/run/maintenance.flag < pg_dump -Fc app_db < migrate --plan < systemctl restart app.service < rm /var/run/maintenance.flag
